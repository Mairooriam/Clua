cmake_minimum_required(VERSION 3.11)
project(Clua C CXX)

# Set C standard to C11
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_definitions(_AMD64_)
# Enable all warnings and sanitizers for GCC/Clang
if(MSVC)
    message(STATUS "MSVC compiler selected: using /W4 warnings")
    add_compile_options(/W4)
    add_compile_options(/external:W0)
else()
    message(STATUS "Using GCC or clang compiler ")
    add_compile_options(
        -Wall
        -Wextra
        -Wpedantic
        -fdiagnostics-color=always
        -Wno-newline-eof
    )
    # Enforce static linking
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static")
endif()

# Use FetchContent to get open62541

## If open62541 sources are already present, use them instead of fetching
set(open62541_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/open62541-src")
if(EXISTS ${open62541_SOURCE_DIR})
message(STATUS "Open62541 Found existing sources at ${open62541_SOURCE_DIR}...")
add_subdirectory(${open62541_SOURCE_DIR} ${CMAKE_BINARY_DIR}/open62541-build)
else()
message(STATUS "Didn't find Open62541 sources at ${open62541_SOURCE_DIR}... Fetching from github")
include(FetchContent)
FetchContent_Declare(
    open62541
        GIT_REPOSITORY https://github.com/open62541/open62541.git
        GIT_TAG v1.5.4
        GIT_SHALLOW ON
        GIT_PROGRESS ON 
    )
    FetchContent_MakeAvailable(open62541)
endif()
find_package(replxx REQUIRED)
find_package(Threads REQUIRED)
 
# Source files
# Add tinycthread.c to sources
set(SOURCES
    src/main.c
    src/log.c
    src/debug_buf.c
    modules/mirua_module.c
    modules/mirua_to_string.c
    modules/mirua_serialization.c
    modules/file/file_windows.c
    src/utils.c
    external/tinycthread.c
    external/sqlite/sqlite3.c
)

# Executable
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
  set_source_files_properties(
    external/sqlite/sqlite3.c
    PROPERTIES COMPILE_OPTIONS "-Wno-unsafe-buffer-usage"
  )
endif()
add_executable(main ${SOURCES})
target_compile_definitions(main PRIVATE LOG_USE_COLOR _CRT_SECURE_NO_WARNINGS)
target_include_directories(main PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/modules
    ${CMAKE_CURRENT_SOURCE_DIR}/external
    ${CMAKE_CURRENT_SOURCE_DIR}/external/sqlite
)
target_link_libraries(main PRIVATE 
open62541 
replxx::replxx 
Threads::Threads 
ws2_32 
iphlpapi
)
