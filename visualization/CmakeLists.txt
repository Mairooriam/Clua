cmake_minimum_required(VERSION 3.11)
set(PROJECT_NAME visualization)
project(Visualization C CXX)

# Set C standard to C11
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_definitions(_AMD64_)
# Enable all warnings and sanitizers for GCC/Clang
if(MSVC)
    message(STATUS "MSVC compiler selected: using /W4 warnings")
    add_compile_options(/W4)
    add_compile_options(/external:W0)
else()
    message(STATUS "Using GCC or clang compiler ")
    add_compile_options(
        -Wall
        -Wextra
        -Wpedantic
        -fdiagnostics-color=always
        -Wno-newline-eof
    )
endif()

find_package(Threads REQUIRED)
find_package(Vulkan REQUIRED)


include(FetchContent)

FetchContent_Declare(
    imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG v1.92.8
    GIT_SHALLOW ON
    GIT_PROGRESS ON
)


FetchContent_MakeAvailable(imgui)

set(IMGUI_SRCS
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_demo.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp
)

FetchContent_Declare(
    sdl
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG release-3.4.10
    GIT_SHALLOW ON
    GIT_PROGRESS ON
)

FetchContent_MakeAvailable(sdl)

set(IMGUI_SRCS
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_demo.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp
)

add_library(imgui STATIC ${IMGUI_SRCS})
target_include_directories(imgui PUBLIC
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends
    ${sdl_SOURCE_DIR}/include
    ${Vulkan_INCLUDE_DIRS}
)
target_link_libraries(imgui PUBLIC SDL3-static ${Vulkan_LIBRARIES})

# Setup ImPlot
set(IMPLOT_SOURCE
    src/implot.cpp
    src/implot_demo.cpp
    src/implot_items.cpp
)
add_library(implot STATIC ${IMPLOT_SOURCE})
add_library(implot::implot ALIAS implot)
target_include_directories(implot PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)
target_link_libraries(implot PUBLIC imgui)
# Source files
# Add tinycthread.c to sources
set(SOURCES
    src/main.cpp
    external/sqlite/sqlite3.c
)

# Executable
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    if(TARGET SDL3-static)
        target_compile_options(SDL3-static PRIVATE -Wno-unsafe-buffer-usage)
    endif()
endif()
add_executable(Vis ${SOURCES})
target_compile_definitions(Vis PRIVATE LOG_USE_COLOR _CRT_SECURE_NO_WARNINGS)
target_include_directories(Vis PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/external
    ${CMAKE_CURRENT_SOURCE_DIR}/external/sqlite
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends
    ${sdl_SOURCE_DIR}/include
    ${Vulkan_INCLUDE_DIRS}
)
target_link_libraries(Vis PRIVATE SDL3-static ${Vulkan_LIBRARIES} imgui implot)
